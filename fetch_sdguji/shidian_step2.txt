def extract_json_from_pre(html):
    """从 HTML 中提取 <pre> 内容并解析 JSON"""
    import re

    m = re.search(r"<pre[^>]*>(.*?)</pre>", html, flags=re.S)
    if not m:
        raise ValueError("没有找到 <pre> JSON 数据")

    json_text = m.group(1).strip()
    return json.loads(json_text)


def save_book_info(bookId, bookName):
    """追加写入 BOOKs.txt"""
    with open(BOOKS_FILE, "a", encoding="utf-8") as f:
        f.write(f"{bookId}\t{bookName}\n")



def save_chapters(bookId, data):
    """保存章节 URL 到 bookId.txt，每行一个 URL，保持原 JSON 顺序"""
    lst = []

    def collect_chapter_ids(node):
        if isinstance(node, dict):
            for k, v in node.items():
                if k == "chapterId":
                    lst.append(str(v))
                else:
                    collect_chapter_ids(v)
        elif isinstance(node, list):
            for item in node:
                collect_chapter_ids(item)

    collect_chapter_ids(data)

    # 拼接 URL
    urls = ["https://www.shidianguji.com/zh/book/" + bookId + "/chapter/" + v for v in lst]

    # 写入文件，每行一个 URL
    out_path = f"{BASE_DIR}/index_{bookId}.txt"
    with open(out_path, "w", encoding="utf-8") as f:
        for url in urls:
            f.write(url + "\n")

    print(f" --  索引文件已經保存: {BASE_DIR}/index_{bookId}.txt")
