def next_chapter(mark_chapter, init_chapters):
    """
    返回 mark_chapter 在列表中的下一個元素，如果沒有返回 None
    """
    try:
        idx = init_chapters.index(mark_chapter)
        if idx + 1 < len(init_chapters):
            return init_chapters[idx + 1]
        else:
            return None  # 已經是最後一個元素
    except ValueError:
        return None  # mark_chapter 不在列表中

def remove_adjacent_duplicates(chapters):
    """
    移除列表中相鄰重複元素，保持原順序
    """
    if not chapters:
        return []

    result = [chapters[0]]  # 第一個元素直接保留
    for item in chapters[1:]:
        if item != result[-1]:
            result.append(item)
    return result

def fetch_chapter_content(page,chapter_url):
    try:
        page.goto(chapter_url, timeout=30000)
        page.wait_for_load_state("networkidle")
        html = page.content()
    except Exception as e:
        print(f"❌ 加载章节失败: {chapter_url} 错误: {e}")
        page.close()
        return None

    page.close()

    soup = BeautifulSoup(html, "html.parser")
    main = soup.find("main", class_="read-layout-main")
    if not main:
        print(f"❌ 未找到章节内容 main: {chapter_url}")
        return None

    p_tags = main.find_all("p")
    if not p_tags:
        print(f"❌ 章节内容为空: {chapter_url}")
        return None

    content = "\n".join([p.get_text(strip=True) for p in p_tags])
    return content
